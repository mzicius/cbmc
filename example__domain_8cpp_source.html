24// Transform updates the domain with the effect of the instruction "from"
25void example_domaint::transform(
26 const irep_idt &function_from,
27 trace_ptrt from,
28 const irep_idt &function_to,
29 trace_ptrt to,
30 ai_baset &ai,
31 const namespacet &ns)
32{
33 std::cerr << "Example domain @ 0x" << this << " transform using instruction "
34 << from->current_location()->location_number << '\n';
35
36 // If e is an exprt (an expression) then
37 // std::cerr << e.pretty()
38 // prints it out
39
40 // Normally how the state of the domain is updated would depend on
41 // what the instruction was (see below). However the state we are
42 // storing is so simple that we can just...
43 this->may_reach = true;
44
45 const goto_programt::instructiont &instruction = *(from->current_location());
46 switch(instruction.type())
47 {
49 case DECL:
50 // Creates a new variable to_code_decl(instruction.code).symbol()
51 break;
52
53 case DEAD:
54 // Says that to_code_dead(instruction.code).symbol() is no longer needed
55 break;
56
57 case ASSIGN:
58 // Set to_code_assign(instruction.code).lhs() to
59 // to_code_assign(instruction.code).rhs()
60 break;
61
62 case GOTO:
63 {
64 // Comparing iterators is safe as the target must be within the same list
65 // of instructions because this is a GOTO.
66 locationt next = from->current_location();
67 next++;
68 if(
69 from->current_location()->get_target() !=
70 next) // If equal then effectively a SKIP
71 {
72 if(next == to->current_location())
73 {
74 // Branch is not taken
75 }
76 else
77 {
78 // Branch is taken
79 }
80 }
81 break;
82 }
83
84 case FUNCTION_CALL:
85 {
86 // Function calls are a bit of a fiddle...
87 // const code_function_callt &code_function_call =
88 // to_code_function_call(instruction.code);
89 break;
90 }
91 /* Removed from more recent versions
92 case RETURN: // Are transformed away into SET_RETURN_VALUE
93 // and then GOTO the end of the function
94 DATA_INVARIANT(false, "Returns must be removed before analysis");
95 break;
96 */
97 case SET_RETURN_VALUE:
98 {
99 // Handle setting the return value
100 break;
101 }
102
103 case ASSUME:
104 // It is safe to over-approximate these by ... ignoring them!
105 break;
106
108 case ASSERT: // An assert is a check; they don't alter execution
109 // If goto-analyzer is run with --verify they will be checked after fixpoint
110 // by using the ai_simplify() method given below.
111 break;
112
113 // Mostly markers
114 case LOCATION: // No action required
115 case SKIP: // No action required
116 case END_FUNCTION: // No action required
117 break;
118
119 case CATCH:
120 case THROW:
121 DATA_INVARIANT(false, "Exceptions must be removed before analysis");
122 break;
123
124 case ATOMIC_BEGIN: // Ignoring is a valid over-approximation
125 case ATOMIC_END: // Ignoring is a valid over-approximation
126 case START_THREAD: // Require a concurrent analysis at higher level
127 case END_THREAD: // Require a concurrent analysis at higher level
128 break;
129
130 case OTHER:
131#if 0
132 DATA_INVARIANT(false, "Unclear what is a safe over-approximation of OTHER");
133#endif
134 break;
135
136 case INCOMPLETE_GOTO:
137 case NO_INSTRUCTION_TYPE:
138 DATA_INVARIANT(false, "Only complete instructions can be analyzed");
139 break;
140 }
141
142 return;
143}

145// Merges the contents of b into this
146bool example_domaint::merge(const example_domaint &b, trace_ptrt, trace_ptrt)
147{
148 std::cerr << "Example domain @ 0x" << this << " merge in example domain @ 0x"
149 << &b << '\n';
150
151 bool old = this->may_reach;
152 this->may_reach |= b.may_reach;
153
154 // Returning true here has consequences!
155 // If merge is true then the domain will be requeued and transformed again.
156 // Fixpoint is reached when all domains return false when merged.
157 return (old != this->may_reach);
158}
