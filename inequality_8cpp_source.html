435 std::shared_ptr<inequality> ineq1,
436 std::shared_ptr<inequality> ineq2)
437{
438 if(ineq1->arity() == 2 && ineq2->arity() == 2)
439 {

444
445 if(dyadic1->x.compare(dyadic2->x) < 0)
446 return true;
447 else if(dyadic1->x.compare(dyadic2->x) > 0)
448 return false;
449 else if(dyadic1->y.compare(dyadic2->y) < 0)
450 return true;
451 else if(dyadic1->y.compare(dyadic2->y) > 0)
452 return false;
453 else
454 return syntax_compare(ineq1, ineq2);
455 }
456 else if(ineq1->arity() == 1 && ineq2->arity() == 2)
457 {
458 std::shared_ptr<unary_inequality> unary =
459 std::dynamic_pointer_cast<unary_inequality>(ineq1);

462
463 if(unary->x.compare(dyadic->x) < 0)
464 return true;

466 return false;
467 else
468 return syntax_compare(unary, dyadic);
469 }
470 else if(ineq1->arity() == 2 && ineq2->arity() == 1)
471 {
472 return !lex_compare(ineq2, ineq1);

598inequality::ordering
599cmp_angle(std::shared_ptr<inequality> a, std::shared_ptr<inequality> b)
600{
601 inequality::ordering res = inequality::EQ;
602 mp_integer a1, b1, a2, b2;
603
604 auto get_ab = [&](std::shared_ptr<inequality> ineq, mp_integer &a, mp_integer &b) {
605 if(cast_to_unary(ineq))
606 {
607 auto u = cast_to_unary(ineq);
608 auto dim_target = dimensions.find(u->x);
609 if(dim_target == dimensions.begin())
610 {
611 a = u->a;
612 b = 0;
613 }
614 else
615 {
616 a = 0;
617 b = u->a;
618 }
619 }
620 else
621 {
622 auto d = cast_to_dyadic(ineq);
623 if(dimensions.find(d->x) == dimensions.begin())
624 {
625 a = d->a;
626 b = d->b;
627 }
628 else
629 {
630 a = d->b;
631 b = d->a;
632 }
633 }
634 };
635
636 get_ab(a, a1, b1);
637 get_ab(b, a2, b2);
638
639 auto calc_dir = [](mp_integer a, mp_integer b) -> inequality::direction {
640 if(a > 0)
641 return (b >= 0) ? inequality::East : inequality::South;
642 else if(a < 0)
643 return (b > 0) ? inequality::North : inequality::West;
644 else
645 return (b > 0) ? inequality::North : inequality::South;
646 };
647
648 inequality::direction dir_a = calc_dir(a1, b1);
649 inequality::direction dir_b = calc_dir(a2, b2);
650
651 if(dir_a == dir_b)
652 {
653 if((a2 * b1) == (a1 * b2))
654 return res;
655 else if((a2 * b1) > (a1 * b2))
656 res = inequality::GT;
657 else
658 res = inequality::LT;
659 }
660 else
661 {
662 if(dir_a > dir_b)
663 res = inequality::GT;
664 else
665 res = inequality::LT;
666 }
667
668 return res;
669}
